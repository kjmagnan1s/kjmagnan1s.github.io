document.addEventListener('DOMContentLoaded', function() {
  // Initialize AOS
  AOS.init({
    duration: 800,
    once: true,
    offset: 100,
    easing: 'ease-out-cubic'
  });

  // View Toggle Functionality
  const staticViewBtn = document.getElementById('staticViewBtn');
  const dashboardViewBtn = document.getElementById('dashboardViewBtn');
  const staticResume = document.getElementById('staticResume');
  const dashboardResume = document.getElementById('dashboardResume');

  staticViewBtn.addEventListener('click', () => {
    staticViewBtn.classList.add('active');
    dashboardViewBtn.classList.remove('active');
    staticResume.style.display = 'block';
    dashboardResume.style.display = 'none';
  });

  dashboardViewBtn.addEventListener('click', () => {
    dashboardViewBtn.classList.add('active');
    staticViewBtn.classList.remove('active');
    dashboardResume.style.display = 'block';
    staticResume.style.display = 'none';
    initializeTableau();
  });

  // Initialize Tableau
  function initializeTableau() {
    var divElement = document.getElementById('viz1745077262291');
    var vizElement = divElement.getElementsByTagName('object')[0];
    vizElement.style.width = '100%';
    vizElement.style.height = '1000px';
    vizElement.style.padding = '0';
    var scriptElement = document.createElement('script');
    scriptElement.src = 'https://public.tableau.com/javascripts/api/viz_v1.js';
    vizElement.parentNode.insertBefore(scriptElement, vizElement);
  }

  // Return to Top Button Functionality
  const returnToTop = document.getElementById('returnToTop');
  
  window.addEventListener('scroll', function() {
    if (window.pageYOffset > 300) {
      returnToTop.style.display = 'block';
    } else {
      returnToTop.style.display = 'none';
    }
  });
  
  returnToTop.addEventListener('click', function() {
    window.scrollTo({
      top: 0,
      behavior: 'smooth'
    });
  });
});
</script>
